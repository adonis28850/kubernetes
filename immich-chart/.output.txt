# Immich Network Policies
# These policies restrict traffic to only what is necessary for Immich to function.
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: immich-server
  namespace: immich
spec:
  podSelector:
    matchLabels:
      app.kubernetes.io/component: server
  policyTypes:
    - Ingress
    - Egress
  ingress:
    # Allow traffic from Gateway
    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: traefik
      ports:
        - protocol: TCP
          port: 3001
  egress:
    # Allow DNS
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53
    # Allow traffic to Database, Valkey and ML
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/component: database
      ports:
        - protocol: TCP
          port: 5432
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/component: valkey
      ports:
        - protocol: TCP
          port: 6379
    - to:
        - podSelector:
            matchLabels:
              app.kubernetes.io/component: machine-learning
      ports:
        - protocol: TCP
          port: 3003

# Source: immich-chart/templates/pv.yaml
# Persistent Volumes for Immich
# Using local storage with static provisioning and Retain policy to preserve data.
apiVersion: v1
kind: PersistentVolume
metadata:
  name: immich-library-pv
spec:
  capacity:
    storage: 100Gi
  volumeMode: Filesystem
  accessModes:
    - ReadWriteOnce
  persistentVolumeReclaimPolicy: Retain
  storageClassName: local-path
  local:
    path: /home/antonio/data/immich/library
  nodeAffinity:
    required:
      nodeSelectorTerms:
        - matchExpressions:
            - key: kubernetes.io/hostname
              operator: In
              values:
                - k3s-cluster
---

# Source: immich-chart/templates/deployment.yaml
# Immich Component Deployments
apiVersion: apps/v1
kind: Deployment
metadata:
  name: immich-server
  namespace: immich
  labels:
    app.kubernetes.io/name: immich
    app.kubernetes.io/component: server
spec:
  replicas: 1
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  selector:
    matchLabels:
      app.kubernetes.io/name: immich
      app.kubernetes.io/component: server
  template:
    metadata:
      labels:
        app.kubernetes.io/name: immich
        app.kubernetes.io/component: server
    spec:
      nodeSelector:
        kubernetes.io/hostname: k3s-cluster
      securityContext:
        fsGroup: 1000
        runAsGroup: 1000
        runAsUser: 1000
      containers:
        - name: server
          image: "ghcr.io/immich-app/immich-server:v2.5.0"
          imagePullPolicy: IfNotPresent
          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
              - ALL
            readOnlyRootFilesystem: true
            runAsNonRoot: true
            runAsUser: 1000
          ports:
            - containerPort: 3001
          env:
            - name: DB_HOSTNAME
              value: immich-database
            - name: DB_USERNAME
              value: postgres
            - name: DB_PASSWORD
              value: immichpassword
            - name: DB_DATABASE_NAME
              value: immich
            - name: REDIS_HOSTNAME
              value: immich-valkey
            - name: IMMICH_MACHINE_LEARNING_URL
              value: http://immich-machine-learning:3003
            - name: IMMICH_METRICS
              value: "false"
          volumeMounts:
            - name: library
              mountPath: /usr/src/app/upload
            - name: tmp
              mountPath: /tmp
            - name: geocoding-dump
              mountPath: /usr/src/app/node_modules/local-reverse-geocoder/data
      volumes:
        - name: library
          persistentVolumeClaim:
            claimName: immich-library
        - name: tmp
          emptyDir: {}
        - name: geocoding-dump
          emptyDir: {}
---

      containers:

          image: "tensorchord/vectorchord:pg14-v0.2.2"
          imagePullPolicy: IfNotPresent
          securityContext:
            # Database needs some specific settings or it might fail with read-only root FS if not careful
            # But let's try the strict one first.

            readOnlyRootFilesystem: true
            runAsNonRoot: true

            - name: POSTGRES_USER
              value: postgres
            - name: POSTGRES_PASSWORD
              value: immichpassword
          ports:
            - containerPort: 5432
          volumeMounts:
            - name: db-data
              mountPath: /var/lib/postgresql/data
            - name: tmp
              mountPath: /tmp
            - name: run-database
              mountPath: /var/run/database
      volumes:
        - name: db-data
          persistentVolumeClaim:
            claimName: immich-database

          emptyDir: {}
