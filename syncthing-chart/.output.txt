# Source: syncthing/templates/networkpolicy.yaml
# NetworkPolicy to restrict traffic to the Syncthing pod
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: syncthing-policy
  namespace: syncthing
spec:
  podSelector:
    matchLabels:
      app: syncthing
  policyTypes:
    - Ingress
    - Egress
  ingress:
    # Allow TCP traffic (GUI and Sync) from the Gateway namespace
    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: traefik
      ports:
        - protocol: TCP
          port: 8384
        - protocol: TCP
          port: 22000
    # Allow UDP traffic (Sync) from the Traefik namespace
    - from:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: traefik
      ports:
        - protocol: UDP
          port: 22000
    # Allow Local Discovery (UDP) from the local network - Not working with Traefik + UDP broadcasts
    #- from:
    #    - ipBlock:
    #        cidr: 0.0.0.0/0
    #  ports:
    #    - protocol: UDP
    #      port: 21027
  egress:
    # Allow DNS resolution
    - to:
        - namespaceSelector:
            matchLabels:
              kubernetes.io/metadata.name: kube-system
      ports:
        - protocol: UDP
          port: 53
        - protocol: TCP
          port: 53
    # Allow outbound traffic for syncing with external nodes, relays, and discovery
    - to:
        - ipBlock:
            cidr: 0.0.0.0/0
      ports:
        - protocol: TCP
          port: 22000
        - protocol: UDP
          port: 22000
        # Local Discovery
        #- protocol: UDP
        #  port: 21027
        # Global Discovery
        - protocol: TCP
          port: 443
        # Relay Protocol
        - protocol: TCP
          port: 22067
        # Relay Connection
        - protocol: TCP
          port: 22070

# Source: syncthing/templates/pv.yaml
# Static PersistentVolume for Syncthing configuration
# Uses local-path provisioner but with Retain reclaim policy and static path
apiVersion: v1
kind: PersistentVolume
metadata:
  name: syncthing-config-pv
spec:
  capacity:
    storage: 1Gi
  volumeMode: Filesystem
  accessModes:
    - ReadWriteOnce
  persistentVolumeReclaimPolicy: Retain
  storageClassName: local-path
  local:
    path: /home/antonio/data/syncthing/config/
  nodeAffinity:
    required:
      nodeSelectorTerms:
        - matchExpressions:
            - key: kubernetes.io/hostname
              operator: In
              values:
                - fedora

# Source: syncthing/templates/deployment.yaml
# Deployment for Syncthing application
apiVersion: apps/v1
kind: Deployment
metadata:
  name: syncthing
  namespace: syncthing
  labels:
    app: syncthing
spec:
  replicas: 1
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0
  selector:
    matchLabels:
      app: syncthing
  template:
    metadata:
      labels:
        app: syncthing
    spec:
      securityContext:
        fsGroup: 1000
        fsGroupChangePolicy: Always
        runAsGroup: 1000
        runAsNonRoot: true
        runAsUser: 1000
        seccompProfile:
          type: RuntimeDefault
      containers:
        - name: syncthing
          image: "syncthing/syncthing:latest"
          imagePullPolicy: Always
          securityContext:
            allowPrivilegeEscalation: false
            capabilities:
              drop:
              - ALL
            readOnlyRootFilesystem: true
          env:
            # Environment variables defined in values.yaml
            - name: STNONATPMP
              value: "1"
            - name: STNOUPNP
              value: "1"
            # Internal Syncthing address and directory configurations
            - name: STGUIADDRESS
              value: "0.0.0.0:8384"
            - name: STHOMEDIR
              value: "/var/syncthing/config"
            - name: HOME
              value: "/var/syncthing"
          ports:
            - name: gui
              containerPort: 8384
              protocol: TCP
            - name: sync-tcp
              containerPort: 22000
              protocol: TCP
            - name: sync-udp
              containerPort: 22000
              protocol: UDP
          livenessProbe:
            httpGet:
              path: /
              port: gui
            initialDelaySeconds: 30
            periodSeconds: 10
          readinessProbe:
            httpGet:
              path: /
              port: gui
            initialDelaySeconds: 5
            periodSeconds: 10
          volumeMounts:
            - name: config
              mountPath: /var/syncthing/config
            - name: data
              mountPath: /var/syncthing/data
      # Deployment uses static PVCs for configuration and data persistence
      volumes:
        - name: config
          persistentVolumeClaim:
            claimName: syncthing-config
        - name: data
          persistentVolumeClaim:
            claimName: syncthing-data
---
# Source: syncthing/templates/gateway.yaml
# Gateway resource that defines the listeners for Syncthing.
# It is automatically created if it doesn't already exist in the cluster.
apiVersion: gateway.networking.k8s.io/v1
kind: Gateway
metadata:
  name: traefik-gateway
  namespace: traefik
spec:
  gatewayClassName: traefik
  listeners:
    - name: websecure
      port: 443
      protocol: HTTPS
      allowedRoutes:
        namespaces:
          from: All
      tls:
        mode: Terminate
        certificateRefs:
          - name: syncthing-tls
    - name: syncthing-tcp
      port: 22000
      protocol: TCP
      allowedRoutes:
        kinds:
          - group: gateway.networking.k8s.io
            kind: TCPRoute
        namespaces:
          from: All
    - name: syncthing-udp
      port: 22000
      protocol: UDP
      allowedRoutes:
        kinds:
          - group: gateway.networking.k8s.io
            kind: UDPRoute
        namespaces:
          from: All
---
# Source: syncthing/templates/httproute.yaml
# This HTTPRoute configures the Syncthing GUI to be accessible via the Gateway API.
# It matches the hostname defined in values.yaml and routes traffic to the syncthing service on the GUI port.
apiVersion: gateway.networking.k8s.io/v1
kind: HTTPRoute
metadata:
  name: syncthing-gui
  namespace: syncthing
spec:
  parentRefs:
    - name: traefik-gateway
      namespace: traefik
  hostnames:
    - syncthing.k3s.local
  rules:
    - matches:
        - path:
            type: PathPrefix
            value: /
      backendRefs:
        - name: syncthing
          port: 8384
---
# Source: syncthing/templates/ingressrouteudp.yaml
# IngressRouteUDP resources for Syncthing UDP traffic.
# This approach is used as a fallback because some Gateway API controllers might not fully support UDPRoute yet.
# DEPRECATION NOTICE: This IngressRouteUDP implementation is intended to be replaced by Gateway API's UDPRoute in the future.
apiVersion: traefik.io/v1alpha1
kind: IngressRouteUDP
metadata:
  name: syncthing-udp
  namespace: syncthing
spec:
  entryPoints:
    - syncthing-udp
  routes:
    - services:
        - name: syncthing
          port: 22000
---
